export const Severity = t.enum("IncidentSeverity", [
  "sev1",
  "sev2",
  "sev3",
  "sev4",
] as const);

export const NotificationChannel = t.enum("NotificationChannel", [
  "email",
  "slack",
  "sms",
] as const);

export const IncidentStatus = gen.variant.stateMachine("IncidentStatus", {
  states: ["open", "acknowledged", "mitigated", "resolved", "cancelled"] as const,
  transitions: {
    open: ["acknowledged", "cancelled"],
    acknowledged: ["mitigated", "resolved", "cancelled"],
    mitigated: ["resolved"],
    resolved: [],
    cancelled: [],
  },
  terminal: ["resolved", "cancelled"] as const,
});

export class Incident extends domain.entity.class("Incident", {
  traits: [traits.audited, traits.syncable],

  merge: merge.entityPolicy("IncidentMergePolicy", {
    default: merge.fieldWise({ conflict: "may_conflict" }),

    fields: {
      status: merge.stateMachine({
        machine: IncidentStatus,
        conflict: merge.manual({
          surface: "IncidentConflictResolutionView",
        }),
        laws: [
          laws.validStateTransition({ assurance: assurance.byConstruction() }),
        ],
      }),

      severity: merge.lastWriteWins({ clock: "server" }),
      summary: merge.lastWriteWins({ clock: "server" }),
      updatedAt: merge.max({ assurance: assurance.byConstruction() }),
    },
  }),

  fields: {
    id: domain.field(t.uuid).primary(),
    organizationId: domain.field(Organization.fields.id.type).indexed(),
    projectId: domain.field(Project.fields.id.type).indexed(),
    title: domain.field(t.string),
    summary: domain.field(t.optional(t.string)),
    severity: domain.field(Severity),
    status: domain.field(IncidentStatus.type),
    createdByUserId: domain.field(User.fields.id.type),
    assignedTeamId: domain.field(t.optional(Team.fields.id.type)),
    acknowledgedByUserId: domain.field(t.optional(User.fields.id.type)),
    acknowledgedAt: domain.field(t.optional(t.datetime)),
    mitigatedAt: domain.field(t.optional(t.datetime)),
    resolvedAt: domain.field(t.optional(t.datetime)),
    cancelledAt: domain.field(t.optional(t.datetime)),
    createdAt: domain.field(t.datetime),
    updatedAt: domain.field(t.datetime),
  },

  invariants: [
    invariant.required("incident.title", {
      field: "title",
      diagnostic: diagnostic.define("incident.title.required", {
        severity: "error",
        message: "Incident title is required.",
      }),
    }),

    invariant.custom("sev1_requires_assigned_team", {
      subject: "Incident",
      when: ({ incident }) => expr.eq(incident.severity, "sev1"),
      must: ({ incident }) => expr.isNotNull(incident.assignedTeamId),
      diagnostic: diagnostic.define("incident.sev1.requires_team", {
        severity: "error",
        audience: ["developer", "ai_agent", "user"],
        message: "SEV1 incidents must have an assigned team.",
        remediation: {
          kind: "set_field",
          field: "assignedTeamId",
        },
      }),
    }),
  ],
}) {}

export const canViewIncident = rule.define("canViewIncident")((ctx, r) =>
  r
    .vars({
      session: AuthSession,
      incident: Incident,
    })
    .when(({ session, incident }) =>
      expr.eq(session.organizationId, incident.organizationId),
    )
    .traits([traits.sqlLowerable]),
);

export const canManageIncident = rule.define("canManageIncident")((ctx, r) =>
  r
    .vars({
      session: AuthSession,
      incident: Incident,
    })
    .when(({ session, incident }) =>
      expr.and(
        expr.eq(session.organizationId, incident.organizationId),
        expr.or(
          expr.eq(session.role, "owner"),
          expr.eq(session.role, "admin"),
          expr.and(
            expr.eq(session.role, "responder"),
            expr.includes(session.teamIds, incident.assignedTeamId),
          ),
        ),
      ),
    )
    .traits([traits.sqlLowerable]),
);

export const canAcknowledgeIncident = rule.define("canAcknowledgeIncident")((ctx, r) =>
  r
    .vars({
      session: AuthSession,
      incident: Incident,
    })
    .when(({ session, incident }) =>
      expr.and(
        canManageIncident.call({ session, incident }),
        expr.eq(incident.status, "open"),
      ),
    ),
);

export const canResolveIncident = rule.define("canResolveIncident")((ctx, r) =>
  r
    .vars({
      session: AuthSession,
      incident: Incident,
    })
    .when(({ session, incident }) =>
      expr.and(
        canManageIncident.call({ session, incident }),
        expr.or(
          expr.eq(incident.status, "acknowledged"),
          expr.eq(incident.status, "mitigated"),
        ),
      ),
    ),
);

export const IncidentPolicy = app.auth.tenantPolicy("IncidentPolicy", {
  tenant: Organization,
  entity: Incident,

  rules: {
    read: canViewIncident,
    update: canManageIncident,

  enforce: {
    read: ["server", "database"],
    update: ["server"],
  },

  exposeClientHints: true,
});

export const SetField = gen.operation.define("SetField", {
  input: t.object("SetFieldInput", {
    field: t.ref("FieldRef"),
    value: t.unknown,
  }),

  output: t.object("SetFieldOutput", {
    previousValue: t.unknown,
    nextValue: t.unknown,
  }),

  traits: [traits.deterministic, traits.patchable, traits.invertible],

  laws: [
    laws.rollbackSafe({ assurance: assurance.byConstruction() }),
  ],

  protocols: {
    patch: gen.operation.patchProtocol({
      producePatch: ({ input }) => gen.patch.setField(input.field, input.value),
      invertPatch: ({ output }) => gen.patch.setField(output.field, output.previousValue),
    }),
  },
});

export const AppendIncidentUpdate = gen.operation.define("AppendIncidentUpdate", {
  input: t.object("AppendIncidentUpdateInput", {
    incidentId: Incident.fields.id.type,
    updateId: IncidentUpdate.fields.id.type,
  }),

  output: t.void,

  traits: [traits.deterministic, traits.appendOnly, traits.offlineReplaySafe],

  laws: [
    laws.idempotent({
      key: ({ input }) => input.updateId,
      assurance: assurance.byConstruction(),
    }),
  ],
});

export const IncidentLifecycle = app.recipes.stateMachineFeature("IncidentLifecycle", {
  entity: Incident,
  state: Incident.fields.status,
  machine: IncidentStatus,
  timestampField: Incident.fields.updatedAt,

  audit: {
    entity: AuditLog,
    actorContext: AuthSession,
  },

  transitionOperation: {
    traits: [traits.deterministic, traits.patchable, traits.serverOnly],
    laws: [
      laws.validStateTransition({ machine: IncidentStatus, assurance: assurance.byConstruction() }),
      laws.rollbackSafe({ assurance: assurance.byConstruction() }),
    ],
  },

  transitions: {
    acknowledge: {
      name: "acknowledgeIncident",
      from: "open",
      to: "acknowledged",

      input: {
        incidentId: Incident.fields.id.type,
      },

      requires: [AuthSession, RequestClock],
      guard: canAcknowledgeIncident,

      set: ({ session, clock }) => ({
        acknowledgedByUserId: session.userId,
        acknowledgedAt: clock.now,
        updatedAt: clock.now,
      }),

      event: {
        name: "IncidentAcknowledged",
        payload: ({ input, session, clock, incident }) => ({
          eventId: app.uuid(),
          organizationId: incident.organizationId,
          incidentId: input.incidentId,
          acknowledgedByUserId: session.userId,
          occurredAt: clock.now,
        }),
      },

      optimistic: "derive",
      offline: true,
    },

    resolve: {
      name: "resolveIncident",
      from: ["acknowledged", "mitigated"],
      to: "resolved",

      input: {
        incidentId: Incident.fields.id.type,
        resolutionSummary: t.string,
      },

      requires: [AuthSession, RequestClock],
      guard: canResolveIncident,

      append: {
        entity: IncidentUpdate,
        operation: AppendIncidentUpdate,
        values: ({ input, session, clock, incident }) => ({
          id: app.uuid(),
          organizationId: incident.organizationId,
          incidentId: input.incidentId,
          authorUserId: session.userId,
          body: input.resolutionSummary,
          createdAt: clock.now,
        }),
      },

      set: ({ clock }) => ({
        resolvedAt: clock.now,
        updatedAt: clock.now,
      }),

      event: {
        name: "IncidentResolved",
        payload: ({ input, session, clock, incident }) => ({
          eventId: app.uuid(),
          organizationId: incident.organizationId,
          incidentId: input.incidentId,
          resolvedByUserId: session.userId,
          occurredAt: clock.now,
        }),
      },

      optimistic: "derive",
      offline: false,
    },
  },
});

      summary: t.optional(t.string),
      severity: Severity,
      assignedTeamId: t.optional(Team.fields.id.type),
    })
    .requires(AuthSession)
    .requires(RequestClock)
    .runsIn(ServerBoundary)
    .writes([Incident, AuditLog])
    .applies([SetField])
    .body(({ input, session, clock }) => {
      const incident = Incident.insert({
        id: app.uuid(),
        organizationId: input.organizationId,
        projectId: input.projectId,
        title: input.title,
        summary: input.summary,
        severity: input.severity,
        status: "open",
        assignedTeamId: input.assignedTeamId,
        createdByUserId: session.userId,
        createdAt: clock.now,
        updatedAt: clock.now,
      });

      AuditLog.insert({
        id: app.uuid(),
        organizationId: input.organizationId,
        actorType: "user",
        actorUserId: session.userId,
        action: "incident.declared",
        entityRef: incident.ref.id,
        payload: { severity: input.severity },
        createdAt: clock.now,
      });

      return { incident };
    })
    .event("IncidentDeclared", ({ input, result, clock }) => ({
      eventId: app.uuid(),
      organizationId: input.organizationId,
      incidentId: result.incident.id,
      projectId: input.projectId,
      severity: input.severity,
      title: input.title,
      occurredAt: clock.now,
    })),
);

export const AtLeastOnceRequiresIdempotency = invariant.family("AtLeastOnceRequiresIdempotency")({
  subject: dispatch.nodes.dispatch,

  when: ({ dispatchNode }) =>
    expr.eq(dispatchNode.delivery.guarantee, "at_least_once"),

  must: ({ dispatchNode }) =>
    expr.existsEdge(dispatch.edges.usesIdempotency, {
      dispatch: dispatchNode,
    }),

  diagnostic: diagnostic.define("dispatch.at_least_once.requires_idempotency", {
    severity: "error",
    audience: ["developer", "ai_agent"],
    message: "Dispatch uses at-least-once delivery but has no idempotency plan.",
    remediation: {
      kind: "add_edge",
      edgeKind: dispatch.edges.usesIdempotency,
      suggestion: "Use idempotency.eventId(...), idempotency.inputHash(...), or idempotency.commandId(...).",
    },
  }),
});

export const ClientCannotReadServerOnlyFields = invariant.family("ClientCannotReadServerOnlyFields")({
  subject: ui.nodes.component,

  when: ({ component }) => expr.eq(component.boundary, ClientBoundary),

  must: ({ component }) =>
    expr.not(
      expr.existsPath(component, {
        via: [ui.edges.usesResource, callable.edges.queryReads],
        toTrait: traits.serverOnly,
      }),
    ),

  diagnostic: diagnostic.define("ui.client.server_only_field", {
    severity: "error",
    audience: ["developer", "ai_agent"],
    message: "Client component reads a server-only field through a query/resource path.",
    remediation: {
      kind: "suggest_projection",
      suggestion: "Use a client-safe projection or move the component to a server boundary.",
    },
  }),
});

export const OfflineWriteRequiresMergePolicy = invariant.family("OfflineWriteRequiresMergePolicy")({
  subject: callable.nodes.action,

  when: ({ action }) => expr.hasTrait(action, traits.offlineReplaySafe),

  must: ({ action }) =>
    expr.everyWrittenField(action, (field) =>
      expr.existsEdge(merge.edges.fieldUsesMergeStrategy, { field }),
    ),

  diagnostic: diagnostic.define("offline.write.requires_merge_policy", {
    severity: "error",
    audience: ["developer", "ai_agent"],
    message: "Offline action writes fields without merge strategies.",
    remediation: {
      kind: "add_merge_strategy",
      suggestion: "Attach merge.stateMachine, merge.lastWriteWins, merge.setUnion, or a manual conflict policy.",
    },
  }),
});

```ts
export const canViewIncident = logic.rule("canViewIncident")((ctx, rule) =>
  rule
    .vars({
      session: AuthSession,
      incident: Incident,
    })
    .when(({ session, incident }) =>
      expr.eq(session.organizationId, incident.organizationId),
    )
    .traits([traits.sqlLowerable]),
);

export const canManageIncident = logic.rule("canManageIncident")((ctx, rule) =>
  rule
    .vars({
      session: AuthSession,
      incident: Incident,
    })
    .when(({ session, incident }) =>
      expr.and(
        expr.eq(session.organizationId, incident.organizationId),
        expr.or(
          expr.eq(session.role, "owner"),
          expr.eq(session.role, "admin"),
          expr.and(
            expr.eq(session.role, "responder"),
            expr.includes(session.teamIds, incident.assignedTeamId),
          ),
        ),
      ),
    )
    .traits([traits.sqlLowerable]),
);

export const canAcknowledgeIncident = logic.rule("canAcknowledgeIncident")(
  (ctx, rule) =>
    rule
      .vars({
        session: AuthSession,
        incident: Incident,
      })
      .when(({ session, incident }) =>
        expr.and(
          canManageIncident.call({ session, incident }),
          expr.eq(incident.status, "open"),
        ),
      ),
);

export const canResolveIncident = logic.rule("canResolveIncident")((ctx, rule) =>
  rule
    .vars({
      session: AuthSession,
      incident: Incident,
    })
    .when(({ session, incident }) =>
      expr.and(
        canManageIncident.call({ session, incident }),
        expr.or(
          expr.eq(incident.status, "acknowledged"),
          expr.eq(incident.status, "mitigated"),

      ),
    ),

        OpenIncidentsKey({
          organizationId: input.organizationId,
          projectId: input.projectId,
        }),
      )
      .output(t.array(Incident.type))
      .body(({ input }) =>
        Incident.where((i) =>
          expr.and(
            expr.eq(i.organizationId, input.organizationId),
            input.projectId
              ? expr.eq(i.projectId, input.projectId)
              : expr.true(),
            expr.not(expr.includes(["resolved", "cancelled"], i.status)),
          ),
        ).orderBy((i) => [i.severity.desc(), i.updatedAt.desc()]),
      ),
);

export const getIncidentDetail = callable.query("getIncidentDetail")(
  (ctx, query) =>
    query
      .input({

      .key(({ input }) =>
        IncidentDetailKey({
          incidentId: input.incidentId,
        }),
      )
      .output(Incident.type)
      .body(({ input }) =>
        Incident.findUnique({
          id: input.incidentId,
        }),
      ),
);

export const getIncidentTimeline = callable.query("getIncidentTimeline")(

export const declareIncident = callable.action("declareIncident")(
  (ctx, action) =>
    action
      .input({
        organizationId: Organization.fields.id.type,
        projectId: Project.fields.id.type,
        title: t.string,
        summary: t.optional(t.string),
        severity: Severity,
        assignedTeamId: t.optional(Team.fields.id.type),
      })
      .requires(AuthSession)
      .requires(RequestClock)
      .runsIn(ServerBoundary)
      .writes([
        Incident,
        AuditLog,
      ])
      .applies([
        SetField,
      ])
      .emits(IncidentDeclared, ({ input, session, clock, result }) => ({
        eventId: gen.uuid(),
        organizationId: input.organizationId,
        incidentId: result.incident.id,
        projectId: input.projectId,
        severity: input.severity,
        title: input.title,
        occurredAt: clock.now,
      }))
      .body(({ input, session, clock }) => {
        const incident = Incident.insert({
          id: gen.uuid(),
          organizationId: input.organizationId,
          projectId: input.projectId,
          title: input.title,
          summary: input.summary,
          severity: input.severity,
          status: "open",
          createdByUserId: session.userId,
          assignedTeamId: input.assignedTeamId,
          acknowledgedByUserId: null,
          acknowledgedAt: null,
          mitigatedAt: null,
          resolvedAt: null,
          cancelledAt: null,
          createdAt: clock.now,
          updatedAt: clock.now,
        });

        const audit = AuditLog.insert({
          id: gen.uuid(),
          organizationId: input.organizationId,
          actorType: "user",
          actorUserId: session.userId,
          action: "incident.declared",
          entityRef: incident.ref.id,
          payload: { severity: input.severity },
          createdAt: clock.now,
        });

        return { incident, audit };
      }),
);

export const acknowledgeIncident = callable.action("acknowledgeIncident")(
  (ctx, action) =>
    action
      .input({
        incidentId: Incident.fields.id.type,
      })
      .requires(AuthSession)
      .requires(RequestClock)
      .runsIn(ServerBoundary)
      .load({
        incident: getIncidentDetail,
      })
      .guard(({ session, incident }) =>
        canAcknowledgeIncident.call({ session, incident }),
      )
      .applies([
        TransitionIncidentStatus,
        SetField,
      ])
      .writes([
        Incident.fields.status,
        Incident.fields.acknowledgedAt,
        Incident.fields.acknowledgedByUserId,
        Incident.fields.updatedAt,
        AuditLog,
      ])
      .emits(IncidentAcknowledged, ({ input, session, clock, incident }) => ({
        eventId: gen.uuid(),
        organizationId: incident.organizationId,
        incidentId: input.incidentId,
        acknowledgedByUserId: session.userId,
        occurredAt: clock.now,
      }))
      .optimistic({
        strategy: "derive_from_operations",
        fallback: "invalidate",
      })
      .body(({ input, session, clock, incident }) => {
        Incident.update(input.incidentId, {
          status: "acknowledged",
          acknowledgedAt: clock.now,
          acknowledgedByUserId: session.userId,
          updatedAt: clock.now,
        });

        AuditLog.insert({
          id: gen.uuid(),
          organizationId: incident.organizationId,
          actorType: "user",
          actorUserId: session.userId,
          action: "incident.acknowledged",
          entityRef: Incident.ref(input.incidentId).id,
          payload: {},
          createdAt: clock.now,
        });
      }),
);

export const resolveIncident = callable.action("resolveIncident")(
  (ctx, action) =>
    action
      .input({
        incidentId: Incident.fields.id.type,
        resolutionSummary: t.string,
      })
      .requires(AuthSession)
      .requires(RequestClock)
      .runsIn(ServerBoundary)
      .load({
        incident: getIncidentDetail,
      })
      .guard(({ session, incident }) =>
        canResolveIncident.call({ session, incident }),
      )
      .applies([
        TransitionIncidentStatus,
        SetField,
        AppendIncidentUpdate,
      ])
      .writes([
        Incident.fields.status,
        Incident.fields.resolvedAt,
        Incident.fields.updatedAt,
        IncidentUpdate,
        AuditLog,
      ])
      .emits(IncidentResolved, ({ input, session, clock, incident }) => ({
        eventId: gen.uuid(),
        organizationId: incident.organizationId,
        incidentId: input.incidentId,
        resolvedByUserId: session.userId,
        occurredAt: clock.now,
      }))
      .optimistic({
        strategy: "derive_from_operations",
        fallback: "invalidate",
      })
      .body(({ input, session, clock, incident }) => {
        Incident.update(input.incidentId, {
          status: "resolved",
          resolvedAt: clock.now,
          updatedAt: clock.now,
        });

        IncidentUpdate.insert({
          id: gen.uuid(),
          organizationId: incident.organizationId,
          incidentId: input.incidentId,
          authorUserId: session.userId,
          body: input.resolutionSummary,
          createdAt: clock.now,
        });

        AuditLog.insert({
          id: gen.uuid(),
          organizationId: incident.organizationId,
          actorType: "user",
          actorUserId: session.userId,
          action: "incident.resolved",
          entityRef: Incident.ref(input.incidentId).id,
          payload: {
            resolutionSummary: input.resolutionSummary,
          },
          createdAt: clock.now,
        });
      }),
);

export const AtLeastOnceRequiresIdempotency =
  invariant.family("AtLeastOnceRequiresIdempotency")({
    subject: dispatch.nodes.dispatch,

    when: ({ dispatchNode }) =>
      expr.eq(dispatchNode.delivery.guarantee, "at_least_once"),

    must: ({ dispatchNode }) =>
      expr.existsEdge(dispatch.edges.usesIdempotency, {
        dispatch: dispatchNode,
      }),

    diagnostic: diagnostic.define("dispatch.at_least_once.requires_idempotency", {
      severity: "error",
      audience: ["developer", "ai_agent"],
      message:
        "Dispatch uses at-least-once delivery but has no idempotency plan.",
      remediation: {
        kind: "add_edge",
        edgeKind: dispatch.edges.usesIdempotency,
        suggestion: "Use idempotency.eventId(...) or idempotency.inputHash(...).",
      },
    }),
  });

export const ClientCannotReadServerOnlyFields =
  invariant.family("ClientCannotReadServerOnlyFields")({
    subject: ui.nodes.component,

    when: ({ component }) => expr.eq(component.boundary, ClientBoundary),

    must: ({ component }) =>
      expr.not(
        expr.existsPath(component, {
          via: [
            ui.edges.usesResource,
            callable.edges.queryReads,
          ],
          toTrait: traits.serverOnly,
        }),
      ),

    diagnostic: diagnostic.define("ui.client.server_only_field", {
      severity: "error",
      audience: ["developer", "ai_agent"],
      message:
        "Client component reads a server-only field through a query/resource path.",
      remediation: {
        kind: "suggest_boundary_projection",
        suggestion:
          "Add a server-safe projection query or mark the field as client-readable.",
      },
    }),
  });

/**
 * From the same graph, Gen2 can derive:
 *
 * - Postgres tables, indexes, foreign keys, RLS policies, outbox table.
 * - Effect server runtime, providers, layers, actions, errors.
 * - React components, hooks, mutations, disabled states, optimistic updates.
 * - JSON-Render catalog/spec for AI-editable UI.
 * - OpenAPI routes for queries/actions.
 * - Alchemy deployment resources: workers, database, secrets, queues.
 * - Cache keys and invalidation edges.
 * - Offline queue and replay plan.
 * - At-least-once idempotency verification.
 * - State-machine transition diagnostics.
 * - Invariant checks and AI repair hints.
 * - Audit docs and property tests.
 */
```

The key thing this example demonstrates is that the public API can feel domain-specific and ergonomic while the compiler only sees graph facts:

```txt
Node(Entity): Incident
Node(Action): acknowledgeIncident
Node(Rule): canAcknowledgeIncident
Node(Event): IncidentAcknowledged
Node(Queue): NotificationQueue
Node(Component): IncidentDashboard

Edge(OwnsField): Incident -> Incident.status
Edge(ActionWrites): acknowledgeIncident -> Incident.status
Edge(ActionEmits): acknowledgeIncident -> IncidentAcknowledged
Edge(GuardedBy): acknowledgeIncident -> canAcknowledgeIncident
Edge(QueryReads): listOpenIncidents -> Incident.status
Edge(InvalidatesKey): acknowledgeIncident -> OpenIncidentsKey
Edge(DispatchTriggeredBy): notifyOnIncidentDeclared -> IncidentDeclared
Edge(BehaviorRunsAction): AcknowledgeBehavior -> acknowledgeIncident
Edge(ComponentReturnsView): IncidentDashboard -> IncidentDashboardView

Trait: Incident.status is state-machine-backed
Trait: TransitionIncidentStatus is patchable
Trait: NotificationQueue is durable
Trait: IncidentDashboard is client-renderable
Trait: IncidentDeclared is serializable
```

That is the “ultimate API” shape: rich authoring surface, tiny graph ontology, strong TypeScript witness layer, runtime graph verification, and target-specific output derived from the same semantic source.


Below is the **revised version** of the same real-world app, assuming the suggested improvements were integrated.

The main difference from the previous example:

```txt id="5rxumh"
Before:
  Manually wire every entity, rule, action, event, mutation, resource,
  dispatch, UI view, storage mapping, and deployment resource.

Now:
  Use kits, feature modules, recipes, projections, generated graph witnesses,
  capability reports, explain/trace APIs, importers, and semantic graph diffs.

Still:
  Everything lowers to the same graph facts.
```

This is still an idealized target API, but it should feel closer to what the library should actually expose.

```ts id="0qtv1n"
export const canViewIncident = rule("canViewIncident")((ctx, r) =>
  r
    .vars({
      session: AuthSession,
      incident: Incident,
    })
    .when(({ session, incident }) =>
      expr.eq(session.organizationId, incident.organizationId),
    )
    .traits([traits.sqlLowerable]),
);

export const canManageIncident = rule("canManageIncident")((ctx, r) =>
  r
    .vars({
      session: AuthSession,
      incident: Incident,
    })
    .when(({ session, incident }) =>
      expr.and(
        expr.eq(session.organizationId, incident.organizationId),
        expr.or(
          expr.eq(session.role, "owner"),
          expr.eq(session.role, "admin"),
          expr.and(
            expr.eq(session.role, "responder"),
            expr.includes(session.teamIds, incident.assignedTeamId),
          ),
        ),
      ),
    )
    .traits([traits.sqlLowerable]),
);

export const canAcknowledgeIncident = rule("canAcknowledgeIncident")((ctx, r) =>
  r
    .vars({
      session: AuthSession,
      incident: Incident,
    })
    .when(({ session, incident }) =>
      expr.and(
        canManageIncident.call({ session, incident }),
        expr.eq(incident.status, "open"),
      ),
    ),
);

export const canResolveIncident = rule("canResolveIncident")((ctx, r) =>
  r
    .vars({
      session: AuthSession,
      incident: Incident,
    })
    .when(({ session, incident }) =>
      expr.and(
        canManageIncident.call({ session, incident }),
        expr.or(
          expr.eq(incident.status, "acknowledged"),
          expr.eq(incident.status, "mitigated"),
        ),
      ),
    ),
);

export const IncidentLifecycle = app.recipes.stateMachineFeature("IncidentLifecycle", {
  entity: Incident,
  state: Incident.fields.status,
  machine: IncidentStatus,

  timestampField: Incident.fields.updatedAt,

  audit: {
    entity: AuditLog,
    actorContext: AuthSession,
  },

  operations: {
    transition: {
      traits: [
        traits.deterministic,
        traits.patchable,
        traits.serverOnly,
      ],

      laws: [
        laws.validStateTransition({
          machine: IncidentStatus,
        }),

        laws.rollbackSafe({
          assurance: "by_construction",
        }),
      ],
    },
  },

  transitions: {
    acknowledge: {
      name: "acknowledgeIncident",
      from: "open",
      to: "acknowledged",

      input: {
        incidentId: Incident.fields.id.type,
      },

      guard: canAcknowledgeIncident,

      set: ({ session, clock }) => ({
        acknowledgedByUserId: session.userId,
        acknowledgedAt: clock.now,
        updatedAt: clock.now,
      }),

      event: {
        name: "IncidentAcknowledged",
        payload: ({ input, session, clock, incident }) => ({
          eventId: app.uuid(),
          organizationId: incident.organizationId,
          incidentId: input.incidentId,
          acknowledgedByUserId: session.userId,
          occurredAt: clock.now,
        }),
      },

      optimistic: true,
      offline: true,
    },

    resolve: {
      name: "resolveIncident",
      from: ["acknowledged", "mitigated"],
      to: "resolved",

      input: {
        incidentId: Incident.fields.id.type,
        resolutionSummary: t.string,
      },

      guard: canResolveIncident,

      append: {
        entity: IncidentUpdate,
        values: ({ input, session, clock, incident }) => ({
          id: app.uuid(),
          organizationId: incident.organizationId,
          incidentId: input.incidentId,
          authorUserId: session.userId,
          body: input.resolutionSummary,
          createdAt: clock.now,
        }),
      },

      set: ({ clock }) => ({
        resolvedAt: clock.now,
        updatedAt: clock.now,
      }),

      event: {
        name: "IncidentResolved",
        payload: ({ input, session, clock, incident }) => ({
          eventId: app.uuid(),
          organizationId: incident.organizationId,
          incidentId: input.incidentId,
          resolvedByUserId: session.userId,
          occurredAt: clock.now,
        }),
      },

      optimistic: true,
      offline: false,
    },
  },
});

      guard: ({ session, input }) =>
        expr.eq(session.organizationId, input.organizationId),

      where: ({ input, row }) =>
        expr.and(
          expr.eq(row.organizationId, input.organizationId),
          input.projectId
            ? expr.eq(row.projectId, input.projectId)
            : expr.true(),
          expr.not(expr.includes(["resolved", "cancelled"], row.status)),
        ),

      orderBy: ({ row }) => [
        row.severity.desc(),
        row.updatedAt.desc(),
      ],
    },

    detail: {
      name: "getIncidentDetail",

      input: {
        incidentId: Incident.fields.id.type,
      },

      key: "incidentDetail",

      requires: [AuthSession],

      whereUnique: ({ input }) => ({
        id: input.incidentId,
      }),

      guardResult: ({ session, result }) =>
        canViewIncident.call({
          session,
          incident: result,
        }),
    },
  },
});

export const AtLeastOnceRequiresIdempotency =
  invariant.family("AtLeastOnceRequiresIdempotency")({
    subject: app.dispatch.nodes.dispatch,

    when: ({ dispatchNode }) =>
      expr.eq(dispatchNode.delivery.guarantee, "at_least_once"),

    must: ({ dispatchNode }) =>
      expr.existsEdge(app.dispatch.edges.usesIdempotency, {
        dispatch: dispatchNode,
      }),

    diagnostic: diagnostic.define("dispatch.at_least_once.requires_idempotency", {
      severity: "error",
      audience: ["developer", "ai_agent"],
      message:
        "Dispatch uses at-least-once delivery but has no idempotency plan.",
      remediation: {
        kind: "add_edge",
        edgeKind: app.dispatch.edges.usesIdempotency,
        suggestion: "Use idempotency.eventId(...) or idempotency.inputHash(...).",
      },
    }),
  });

export const ClientCannotReadServerOnlyFields =
  invariant.family("ClientCannotReadServerOnlyFields")({
    subject: ui.nodes.component,

    when: ({ component }) =>
      expr.eq(component.boundary, ClientBoundary),

    must: ({ component }) =>
      expr.not(
        expr.existsPath(component, {
          via: [
            ui.edges.usesResource,
            app.callable.edges.queryReads,
          ],
          toTrait: traits.serverOnly,
        }),
      ),

    diagnostic: diagnostic.define("ui.client.server_only_field", {
      severity: "error",
      audience: ["developer", "ai_agent"],
      message:
        "Client component reads a server-only field through a query/resource path.",
      remediation: {
        kind: "suggest_projection",
        suggestion:
          "Use a client-safe projection or move the component to a server boundary.",
      },
    }),
  });
```

## Why this version is better

The earlier example proved the graph can express everything. This version shows how the library can be used comfortably.

The important improvements are:

```txt id="mefmpi"
1. Kits establish app-level defaults.
2. Modules provide human/agent edit boundaries.
3. Recipes encode common graph patterns.
4. Projections make client safety explicit.
5. Generated pieces remain addressable and customizable.
6. Capability reports explain target support before emission.
7. Explain/trace APIs make derived facts trustworthy.
8. Importers allow AI-generated and legacy artifacts to become graph facts.
9. Graph diff supports migrations, reviews, and deployment planning.
10. Target legalization is mandatory before emission.
11. Tests/docs/graph witnesses are first-class artifacts.
```

The ultimate API should support both levels:

```txt id="aivzkf"
High-level:
  app.recipes.stateMachineFeature(...)
  app.recipes.dashboard(...)
  app.recipes.notificationPipeline(...)

Low-level:
  domain.entity(...)
  rule(...)
  action(...)
  edge(...)
  dispatch(...)
  ui.attachBehavior(...)
```

That gives users and AI agents a good default path without hiding the semantic graph underneath.
